cmake_minimum_required(VERSION 3.20)
project(pocket_pet VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(pet_core STATIC src/game.cpp src/save.cpp src/sprite.cpp src/animation.cpp src/input.cpp src/render.cpp src/device.cpp)
target_include_directories(pet_core PUBLIC include)
set_target_properties(pet_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(pet_bridge SHARED desktop/bridge.cpp)
target_link_libraries(pet_bridge PRIVATE pet_core)
add_executable(pet_tests tests/core_tests.cpp)
target_link_libraries(pet_tests PRIVATE pet_core)
foreach(target pet_core pet_bridge pet_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME pet_core_tests COMMAND pet_tests)
